export const action = async ({ request }) => {
  const { session } = await authenticate.admin(request);
  const shop = session.shop;
  const method = request.method;

  if (method === "DELETE") {
    const id = new URL(request.url).searchParams.get("id");
    if (!id) return data({ error: "Missing review id" }, { status: 400 });

    const review = await db.review.findFirst({
      where: { id, shop },
      select: { id: true, productRefId: true, media: true },
    });
    if (!review) return data({ error: "Review not found" }, { status: 404 });

    if (Array.isArray(review.media)) {
      for (const item of review.media) {
        if (!item?.publicId) continue;
        const resourceType = item.type === "VIDEO" ? "video" : "image";
        try {
          await cloudinary.uploader.destroy(item.publicId, {
            resource_type: resourceType,
            invalidate: true,
          });
        } catch (err) {
          console.error("[cloudinary delete]", item.publicId, err);
        }
      }
    }

    await db.review.delete({ where: { id } });
    await recomputeProductRatings(review.productRefId, shop);
    return data({ success: true });
  }

  if (method === "PUT") {
    const { id, action: reviewAction, merchantReply } = await request.json();
    if (!id) return data({ error: "Missing review id" }, { status: 400 });

    const existing = await db.review.findFirst({
      where: { id, shop },
      select: { productRefId: true, status: true, publishedAt: true },
    });
    if (!existing) return data({ error: "Review not found" }, { status: 404 });

    const patch = {};
    switch (reviewAction) {
      case "approve":
        patch.status = "PUBLISHED";
        if (!existing.publishedAt) patch.publishedAt = new Date();
        break;
      case "unpublish":
        patch.status = "UNPUBLISHED";
        break;
      case "reject":
        patch.status = "REJECTED";
        break;
      case "feature":
        patch.status = "FEATURED";
        if (!existing.publishedAt) patch.publishedAt = new Date();
        break;
      case "unfeature":
        patch.status = "PUBLISHED";
        break;
      case "reply":
        if (typeof merchantReply !== "string" || merchantReply.trim() === "") {
          return data({ error: "merchantReply is required" }, { status: 400 });
        }
        patch.merchantReply = merchantReply.trim().slice(0, 2000);
        patch.repliedAt = new Date();
        break;
      default:
        return data({ error: "Unknown action" }, { status: 400 });
    }

    const updated = await db.review.update({ where: { id }, data: patch });

    if (["approve", "unpublish", "reject", "feature", "unfeature"].includes(reviewAction)) {
      await recomputeProductRatings(existing.productRefId, shop);
    }

    return data({ success: true, review: updated });
  }

  return data({ error: "Method not allowed" }, { status: 405 });
};

export const action = async ({ request }) => {
  if (request.method !== "POST") {
    return data({ error: "Method not allowed" }, { status: 405 });
  }

  const formData = await request.formData();

  const shop = formData.get("shop");
  const productId = formData.get("productId");
  const authorName = formData.get("authorName");
  const authorEmail = formData.get("authorEmail");
  const title = formData.get("title");
  const body = formData.get("body");
  const ratingRaw = formData.get("rating");
  const mediaRaw = formData.get("media");
  const honeypot = formData.get("website");

  if (honeypot) return data({ success: true });

  if (!shop || !productId || !authorName || !body || !ratingRaw) {
    return data({ error: "Missing required fields" }, { status: 400 });
  }

  if (String(authorName).length > 80) return data({ error: "Name too long" }, { status: 400 });
  if (String(body).length > 5000) return data({ error: "Body too long" }, { status: 400 });
  if (title && String(title).length > 120) return data({ error: "Title too long" }, { status: 400 });

  if (authorEmail && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(String(authorEmail))) {
    return data({ error: "Invalid email" }, { status: 400 });
  }

  const rating = Number(ratingRaw);
  if (!Number.isInteger(rating) || rating < 1 || rating > 5) {
    return data({ error: "Rating must be 1–5" }, { status: 400 });
  }

  let product = await db.product.findUnique({
    where: { shop_productId: { shop, productId: String(productId) } },
    select: { id: true },
  });

  if (!product) {
    // If the product hasn't been synced to our DB yet, create a stub
    // The webhook will eventually update it with the correct title
    product = await db.product.create({
      data: {
        shop,
        productId: String(productId),
        title: "Unknown Product",
      },
      select: { id: true },
    });
  }

  if (authorEmail) {
    const recent = await db.review.findFirst({
      where: {
        shop,
        productId: String(productId),
        authorEmail: String(authorEmail),
        createdAt: { gte: new Date(Date.now() - 24 * 60 * 60 * 1000) },
      },
      select: { id: true },
    });
    if (recent) {
      return data({ error: "You've already reviewed this product recently." }, { status: 429 });
    }
  }

  let verifiedPurchase = false;
  if (authorEmail) {
    try {
      verifiedPurchase = await verifyPurchase({
        shop,
        email: String(authorEmail),
        productId: String(productId),
      });
    } catch (err) {
      console.error("[api.reviews] verifyPurchase failed:", err);
    }
  }

  let media = null;
  if (mediaRaw) {
    try {
      const parsed = JSON.parse(mediaRaw);
      if (Array.isArray(parsed)) {
        const cloudBase = `https://res.cloudinary.com/${process.env.CLOUDINARY_CLOUD_NAME}/`;
        media = parsed
          .filter((m) => m && typeof m.url === "string" && m.url.startsWith(cloudBase))
          .slice(0, MAX_MEDIA_ITEMS)
          .map((m) => ({
            url: m.url,
            type: m.type === "VIDEO" ? "VIDEO" : "IMAGE",
            publicId: m.publicId || null,
          }));
        if (media.length === 0) media = null;
      }
    } catch {
      media = null;
    }
  }

  const review = await db.review.create({
    data: {
      shop,
      productRefId: product.id,
      productId: String(productId),
      authorName: String(authorName).slice(0, 80),
      authorEmail: authorEmail ? String(authorEmail) : null,
      title: title ? String(title).slice(0, 120) : null,
      body: String(body).slice(0, 5000),
      rating,
      status: "PUBLISHED", // Set to PUBLISHED so it displays immediately
      verifiedPurchase,
      media,
      source: "STORE",
      publishedAt: new Date(),
    },
  });

  return data({
    success: true,
    reviewId: review.id,
    message: "Review submitted successfully.",
  });
};

import { data } from "react-router";
import crypto from "node:crypto";
import cloudinary, { CLOUDINARY_FOLDER } from "../utils/cloudinary.server";

const ALLOWED = {
  image: ["jpg", "jpeg", "png", "webp", "gif", "avif"],
  video: ["mp4", "mov", "webm", "m4v"],
};

export const action = async ({ request }) => {
  if (request.method !== "POST") {
    return data({ error: "Method not allowed" }, { status: 405 });
  }

  const { filename, contentType, size } = await request.json();
  if (!filename || !contentType) {
    return data({ error: "filename and contentType are required" }, { status: 400 });
  }

  const resourceType = contentType.startsWith("video/") ? "video" : "image";
  const ext = filename.split(".").pop()?.toLowerCase() ?? "";
  if (!ALLOWED[resourceType].includes(ext)) {
    return data({ error: `Unsupported .${ext}` }, { status: 400 });
  }

  const maxSize = resourceType === "video" ? 50 * 1024 * 1024 : 10 * 1024 * 1024;
  if (typeof size === "number" && size > maxSize) {
    return data({ error: "File too large" }, { status: 400 });
  }
  const publicId = crypto.randomUUID();
  const timestamp = Math.round(Date.now() / 1000);

  const paramsToSign = {
    timestamp,
    public_id: publicId,
    folder: CLOUDINARY_FOLDER,
  };

  const signature = cloudinary.utils.api_sign_request(
    paramsToSign,
    process.env.CLOUDINARY_API_SECRET
  );

  return data({
    cloudName: process.env.CLOUDINARY_CLOUD_NAME,
    apiKey: process.env.CLOUDINARY_API_KEY,
    timestamp,
    publicId,
    folder: CLOUDINARY_FOLDER,
    resourceType,
    signature,
    uploadUrl: `https://api.cloudinary.com/v1_1/${process.env.CLOUDINARY_CLOUD_NAME}/${resourceType}/upload`,
  });
};

    include: { product: true },
    take: 100,
  });

  // Find products that have "Unknown Product" as title
  const unknownProductIds = [...new Set(
    reviews
      .filter((r) => r.product && r.product.title === "Unknown Product")
      .map((r) => r.product.productId)
  )];

  if (unknownProductIds.length > 0 && admin) {
    try {
      const gids = unknownProductIds.map((id) => `"gid://shopify/Product/${id}"`).join(", ");
      const response = await admin.graphql(`
        query {
          nodes(ids: [${gids}]) {
            ... on Product {
              id
              title
            }
          }
        }
      `);
      const data = await response.json();
      
      const updates = [];
      if (data?.data?.nodes) {
        for (const node of data.data.nodes) {
          if (node && node.id && node.title) {
            const numericId = node.id.split("/").pop();
            updates.push(
              db.product.updateMany({
                where: { shop, productId: numericId },
                data: { title: node.title },
              })
            );
            // Update in-memory for immediate display
            reviews.forEach((r) => {
              if (r.product && r.product.productId === numericId) {
                r.product.title = node.title;
              }
            });
          }
        }
        if (updates.length > 0) {
          await Promise.all(updates);
        }
      }
    } catch (err) {
      console.error("Failed to fetch product titles from Shopify", err);
    }
  }

  return { reviews };
};

import db from "../db.server";

export async function bulkSyncProducts(admin, shop) {
  let cursor = null;
  let hasNext = true;

  while (hasNext) {
    const res = await admin.graphql(
      `#graphql
        query GetProducts($cursor: String) {
          products(first: 50, after: $cursor) {
            edges {
              cursor
              node {
                id title handle descriptionHtml vendor productType status
                images(first: 1) { edges { node { url } } }
              }
            }
            pageInfo { hasNextPage }
          }
        }`,
      { variables: { cursor } }
    );
    const json = await res.json();
    const edges = json.data.products.edges;

    for (const { node } of edges) {
      const numericId = node.id.split("/").pop();
      await db.product.upsert({
        where:  { shop_productId: { shop, productId: numericId } },
        create: {
          shop,
          productId: numericId,
          title: node.title,
          handle: node.handle,
          description: node.descriptionHtml,
          vendor: node.vendor,
          productType: node.productType,
          imageUrl: node.images.edges[0]?.node?.url || null,
          status: node.status,
          syncedAt: new Date(),
        },
        update: {
          title: node.title,
          handle: node.handle,
          description: node.descriptionHtml,
          vendor: node.vendor,
          productType: node.productType,
          imageUrl: node.images.edges[0]?.node?.url || null,
          status: node.status,
          syncedAt: new Date(),
        },
      });
    }

    hasNext = json.data.products.pageInfo.hasNextPage;
    cursor = edges[edges.length - 1]?.cursor ?? null;
  }
}

import db from "../db.server";

export async function recomputeProductRatings(productRefId, shop) {
  const agg = await db.review.aggregate({
    where: {
      productRefId,
      shop,
      status: { in: ["PUBLISHED", "FEATURED"] },
    },
    _avg:   { rating: true },
    _count: { _all: true },
  });

  const publishedReviewCount = agg?._count?._all ?? 0;
  const averageRating = publishedReviewCount > 0
    ? Number((agg._avg.rating ?? 0).toFixed(2))
    : 0;

  await db.product.update({
    where: { id: productRefId },
    data: { averageRating, publishedReviewCount, reviewCount: publishedReviewCount },
  });
}

export async function recomputeProductRatingsForReview(reviewId, shop) {
  const review = await db.review.findFirst({
    where: { id: reviewId, shop },
    select: { productRefId: true },
  });
  if (review) await recomputeProductRatings(review.productRefId, shop);
}

    const query = `email:"${safeEmail}" product_id:${safePid} financial_status:paid`;

    const response = await admin.graphql(
        `#graphql
      query VerifyPurchase($query: String!) {
        orders(first: 1, query: $query) {
          edges { node { id name createdAt } }
        }
      }
    `,
        { variables: { query } }
    );

    const parsed = await response.json();
    return (parsed?.data?.orders?.edges?.length ?? 0) > 0;
}

import { PassThrough } from "stream";
import { renderToPipeableStream } from "react-dom/server";
import { ServerRouter } from "react-router";
import { createReadableStreamFromReadable } from "@react-router/node";
import { isbot } from "isbot";
import { addDocumentResponseHeaders } from "./shopify.server";

export const streamTimeout = 5000;

export default async function handleRequest(
  request,
  responseStatusCode,
  responseHeaders,
  reactRouterContext,
) {
  addDocumentResponseHeaders(request, responseHeaders);
  const userAgent = request.headers.get("user-agent");
  const callbackName = isbot(userAgent ?? "") ? "onAllReady" : "onShellReady";

  return new Promise((resolve, reject) => {
    const { pipe, abort } = renderToPipeableStream(
      <ServerRouter context={reactRouterContext} url={request.url} />,
      {
        [callbackName]: () => {
          const body = new PassThrough();
          const stream = createReadableStreamFromReadable(body);

          responseHeaders.set("Content-Type", "text/html");
          resolve(
            new Response(stream, {
              headers: responseHeaders,
              status: responseStatusCode,
            }),
          );
          pipe(body);
        },
        onShellError(error) {
          reject(error);
        },
        onError(error) {
          responseStatusCode = 500;
          console.error(error);
        },
      },
    );

    setTimeout(abort, streamTimeout + 1000);
  });
}

async function uploadReviewMedia(file, appUrl) {
  const presignRes = await fetch(`${appUrl}/api/uploads/presign`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      filename: file.name,
      contentType: file.type,
      size: file.size,
    }),
  });

  if (!presignRes.ok) throw new Error("Failed to get upload URL");
  const { uploadUrl, publicUrl } = await presignRes.json();

  const putRes = await fetch(uploadUrl, {
    method: "PUT",
    headers: { "Content-Type": file.type },
    body: file,
  });

  if (!putRes.ok) throw new Error("Upload failed");
  return publicUrl;
}

                const pct = Math.round((1 + diff) * 100);
                html +=
                    '<span class="rw-star rw-star--half" aria-hidden="true">' +
                    '<span class="rw-star__bg">★</span>' +
                    '<span class="rw-star__fg" style="width:' + pct + '%">★</span>' +
                    '</span>';
            } else {
                html += '<span class="rw-star rw-star--empty" aria-hidden="true">★</span>';
            }
        }
        return html + '</div>';
    };

    async function fetchReviews({ appUrl, shop, productId, page, perPage, sort }) {
        const res = await fetch(
            apiUrl(appUrl, '/api/reviews', { shop, productId, page, perPage, sort }),
            { headers: { Accept: 'application/json' } }
        );
        if (!res.ok) throw new Error('Failed to load reviews (' + res.status + ')');
        return res.json();
    }

    async function uploadReviewMedia(file, appUrl) {
        const presign = await fetch(apiUrl(appUrl, '/api/uploads/presign'), {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                filename: file.name,
                contentType: file.type,
                size: file.size,
            }),
        });
        if (!presign.ok) {
            const e = await presign.json().catch(() => ({}));
            throw new Error(e.error || 'Could not prepare upload');
        }
        const p = await presign.json();

        const form = new FormData();
        form.append('file', file);
        form.append('api_key', p.apiKey);
        form.append('timestamp', String(p.timestamp));
        form.append('public_id', p.publicId);
        form.append('folder', p.folder);
        form.append('signature', p.signature);

        const upload = await fetch(p.uploadUrl, { method: 'POST', body: form });
        if (!upload.ok) throw new Error('Upload failed');
        const data = await upload.json();

        return {
            url: data.secure_url,
            type: p.resourceType === 'video' ? 'VIDEO' : 'IMAGE',
            publicId: data.public_id,
        };
    }

    function renderReview(r, showVerified) {
        const media = Array.isArray(r.media) ? r.media : [];
        const mediaHtml = media.length
            ? '<div class="rw-review__media">' +
              media
                  .filter((m) => m && m.url)
                  .map((m) => {
                      const safe = escapeHtml(m.url);
                      if (m.type === 'VIDEO') {
                          return '<video src="' + safe + '" controls preload="metadata" width="120" height="120"></video>';
                      }
                      return (
                          '<a href="' + safe + '" target="_blank" rel="noopener">' +
                          '<img src="' + safe + '" alt="Review media" loading="lazy"></a>'
                      );
                  })
                  .join('') +
              '</div>'
            : '';

        const verified =
            showVerified && r.verifiedPurchase
                ? '<span class="rw-review__badge">✔ Verified purchase</span>'
                : '';

        const reply = r.merchantReply
            ? '<div class="rw-review__reply"><strong>Store reply</strong>' +
              escapeHtml(r.merchantReply) + '</div>'
            : '';

        return (
            '<article class="rw-review">' +
            '<header class="rw-review__head">' +
            STARS(r.rating) +
            '<span class="rw-review__author">' + escapeHtml(r.authorName || 'Anonymous') + '</span>' +
            verified +
            '<span class="rw-review__date">' + fmtDate(r.publishedAt || r.createdAt) + '</span>' +
            '</header>' +
            (r.title ? '<h4 class="rw-review__title">' + escapeHtml(r.title) + '</h4>' : '') +
            '<p class="rw-review__body">' + escapeHtml(r.body) + '</p>' +
            mediaHtml +
            reply +
            '</article>'
        );
    }

    function renderPagination(container, page, totalPages, onGo) {
        container.innerHTML = '';
        if (totalPages <= 1) return;

        const makeBtn = (label, targetPage, opts = {}) => {
            const b = document.createElement('button');
            b.type = 'button';
            b.textContent = label;
            if (opts.current) b.setAttribute('aria-current', 'page');
            if (opts.disabled) b.disabled = true;
            b.addEventListener('click', () => onGo(targetPage));
            return b;
        };
        const dots = () => {
            const d = document.createElement('span');
            d.textContent = '…';
            d.style.alignSelf = 'center';
            return d;
        };

        container.appendChild(makeBtn('‹', page - 1, { disabled: page <= 1 }));

        const win = 2;
        const start = Math.max(1, page - win);
        const end = Math.min(totalPages, page + win);

        if (start > 1) {
            container.appendChild(makeBtn('1', 1, { current: page === 1 }));
            if (start > 2) container.appendChild(dots());
        }
        for (let p = start; p <= end; p++) {
            container.appendChild(makeBtn(String(p), p, { current: p === page }));
        }
        if (end < totalPages) {
            if (end < totalPages - 1) container.appendChild(dots());
            container.appendChild(makeBtn(String(totalPages), totalPages, { current: page === totalPages }));
        }

        container.appendChild(makeBtn('›', page + 1, { disabled: page >= totalPages }));
    }

        form.addEventListener('submit', async (e) => {
            e.preventDefault();

            const submitBtn = form.querySelector('button[type="submit"]');

            if (!rating) {
                msg.textContent = 'Please select a star rating.';
                return;
            }

            const files = [...form.querySelector('[data-rw-files]').files].slice(0, 5);
            submitBtn.disabled = true;
            msg.textContent = files.length ? 'Uploading media…' : 'Submitting…';

            try {
                const media = [];
                for (const f of files) media.push(await uploadReviewMedia(f, appUrl));

                const fd = new FormData(form);
                fd.delete('files');
                fd.set('shop', shop);
                fd.set('productId', productId);
                fd.set('rating', String(rating));
                if (media.length) fd.set('media', JSON.stringify(media));

                const res = await fetch(apiUrl(appUrl, '/api/reviews'), {
                    method: 'POST',

                    '<p class="rw-form__done">' +

                console.error('[reviews-widget:submit]', err);
                msg.textContent = err.message || 'Something went wrong. Please try again.';
                submitBtn.disabled = false;
            }
        });

        anchorEl.insertAdjacentElement('beforebegin', form);
        form.querySelector('input[name="authorName"]').focus();
    }

    async function initWidget(root) {
        if (root.dataset.rwInit === '1') return;
        root.dataset.rwInit = '1';

        const appUrl = root.dataset.appUrl || DEFAULT_APP_URL;
        const shop = root.dataset.shop;
        const productId = root.dataset.productId;
        const perPage = parseInt(root.dataset.perPage || '5', 10);
        const showVerified = root.dataset.showVerified !== 'false';

        const listEl = root.querySelector('[data-rw-list]');
        const sortEl = root.querySelector('[data-rw-sort]');
        const pagEl = root.querySelector('[data-rw-pagination]');
        const summaryAverage = root.querySelector('[data-rw-average]');
        const summaryStars = root.querySelector('[data-rw-summary-stars]');
        const summaryCount = root.querySelector('[data-rw-count]');
        const writeBtn = root.querySelector('[data-rw-write]');

        let page = 1;
        let sort = 'recent';

        async function load() {
            listEl.innerHTML = '<div class="rw-loading">Loading reviews…</div>';
            try {
                const data = await fetchReviews({ appUrl, shop, productId, page, perPage, sort });

                const reviews = data.reviews || [];
                const total = data.total || 0;
                const average = data.average || 0;

                if (summaryAverage) summaryAverage.textContent = average.toFixed(1);
                if (summaryStars) summaryStars.innerHTML = STARS(average);
                if (summaryCount)
                    summaryCount.textContent = '(' + total + (total === 1 ? ' review' : ' reviews') + ')';

                const totalPages = Math.max(1, Math.ceil(total / perPage));

                if (!reviews.length) {
                    listEl.innerHTML = '<div class="rw-empty">No reviews yet. Be the first!</div>';
                    pagEl.innerHTML = '';
                    return;
                }

                listEl.innerHTML = reviews.map((r) => renderReview(r, showVerified)).join('');
                renderPagination(pagEl, page, totalPages, (p) => {
                    page = p;
                    load();
                    root.scrollIntoView({ behavior: 'smooth', block: 'start' });
                });
            } catch (err) {
                console.error('[reviews-widget]', err);
                listEl.innerHTML = '<div class="rw-empty">Unable to load reviews right now.</div>';
            }
        }

        if (sortEl) {
            sortEl.addEventListener('change', () => {
                sort = sortEl.value;
                page = 1;
                load();
            });
        }

        if (writeBtn) {
            writeBtn.addEventListener('click', () =>
                openWriteForm(root, { appUrl, shop, productId, anchorEl: writeBtn })
            );
        }

        load();
    }

    async function initInline(root) {
        if (root.dataset.rwInit === '1') return;
        root.dataset.rwInit = '1';

        const appUrl = root.dataset.appUrl || DEFAULT_APP_URL;
        const shop = root.dataset.shop;
        const productId = root.dataset.productId;
        const starsEl = root.querySelector('[data-rw-inline-stars]');
        const countEl = root.querySelector('[data-rw-inline-count]');

        try {
            const res = await fetch(apiUrl(appUrl, '/api/reviews/stats', { shop, productId }), {
                headers: { Accept: 'application/json' },
            });
            if (!res.ok) throw new Error(res.status);
            const { average = 0, count = 0 } = await res.json();

            if (starsEl) starsEl.innerHTML = STARS(average);
            if (countEl) countEl.textContent = '(' + count + (count === 1 ? ' review' : ' reviews') + ')';
        } catch (err) {
            console.warn('[reviews-widget:inline]', err);
        }
    }

    function boot(scope) {
        if (!window.ReviewlyAppEnabled) return;

        const widgets = (scope || document).querySelectorAll('.reviews-widget');
        widgets.forEach((el) => {
            el.style.display = '';
            initWidget(el);
        });

        const inlines = (scope || document).querySelectorAll('.rw-inline-rating');
        inlines.forEach((el) => {
            el.style.display = '';
            initInline(el);
        });
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', () => boot());
    } else {
        boot();
    }

    document.addEventListener('shopify:section:load', (e) => boot(e.target));
})();

import fs from "fs";
import { ApiVersion } from "@shopify/shopify-app-react-router/server";
import { shopifyApiProject, ApiType } from "@shopify/api-codegen-preset";
function getConfig() {
  const config = {
    projects: {
      default: shopifyApiProject({
        apiType: ApiType.Admin,
        apiVersion: ApiVersion.July26,
        documents: [
          "./app/**/*.{js,ts,jsx,tsx}",
          "./app/.server/**/*.{js,ts,jsx,tsx}",
        ],
        outputDir: "./app/types",
      }),
    },
  };
  let extensions = [];
  try {
    extensions = fs.readdirSync("./extensions");
  } catch {
  }
  for (const entry of extensions) {
    const extensionPath = `./extensions/${entry}`;
    const schema = `${extensionPath}/schema.graphql`;
    if (!fs.existsSync(schema)) {
      continue;
    }
    config.projects[entry] = {
      schema,
      documents: [`${extensionPath}/**/*.graphql`],
    };
  }
  return config;
}
const config = getConfig();
export default config;

# Shopify App Template - React Router

This is a template for building a [Shopify app](https://shopify.dev/docs/apps/getting-started) using [React Router](https://reactrouter.com/). It was forked from the [Shopify Remix app template](https://github.com/Shopify/shopify-app-template-remix) and converted to React Router.

Rather than cloning this repo, follow the [Quick Start steps](https://github.com/Shopify/shopify-app-template-react-router#quick-start).

Visit the [`shopify.dev` documentation](https://shopify.dev/docs/api/shopify-app-react-router) for more details on the React Router app package.

## Upgrading from Remix

If you have an existing Remix app that you want to upgrade to React Router, please follow the [upgrade guide](https://github.com/Shopify/shopify-app-template-react-router/wiki/Upgrading-from-Remix). Otherwise, please follow the quick start guide below.

## Quick start

### Prerequisites

Before you begin, you'll need to [download and install the Shopify CLI](https://shopify.dev/docs/apps/tools/cli/getting-started) if you haven't already.

### Setup

```shell
shopify app init --template=https://github.com/Shopify/shopify-app-template-react-router
```

### Local Development

```shell
shopify app dev
```

Press P to open the URL to your app. Once you click install, you can start development.

Local development is powered by [the Shopify CLI](https://shopify.dev/docs/apps/tools/cli). It logs into your account, connects to an app, provides environment variables, updates remote config, creates a tunnel and provides commands to generate extensions.

### Authenticating and querying data

To authenticate and query data you can use the `shopify` const that is exported from `/app/shopify.server.js`:

```js
export async function loader({ request }) {
  const { admin } = await shopify.authenticate.admin(request);

  const response = await admin.graphql(`
    {
      products(first: 25) {
        nodes {
          title
          description
        }
      }
    }`);

  const {
    data: {
      products: { nodes },
    },
  } = await response.json();

  return nodes;
}
```

This template comes pre-configured with examples of:

1. Setting up your Shopify app in [/app/shopify.server.ts](https://github.com/Shopify/shopify-app-template-react-router/blob/main/app/shopify.server.ts)
2. Querying data using Graphql. Please see: [/app/routes/app.\_index.tsx](https://github.com/Shopify/shopify-app-template-react-router/blob/main/app/routes/app._index.tsx).
3. Responding to webhooks. Please see [/app/routes/webhooks.tsx](https://github.com/Shopify/shopify-app-template-react-router/blob/main/app/routes/webhooks.app.uninstalled.tsx).
4. Using metafields, metaobjects, and declarative custom data definitions. Please see [/app/routes/app.\_index.tsx](https://github.com/Shopify/shopify-app-template-react-router/blob/main/app/routes/app._index.tsx) and [shopify.app.toml](https://github.com/Shopify/shopify-app-template-react-router/blob/main/shopify.app.toml).

Please read the [documentation for @shopify/shopify-app-react-router](https://shopify.dev/docs/api/shopify-app-react-router) to see what other API's are available.

## Shopify Dev MCP

This template is configured with the Shopify Dev MCP. This instructs [Cursor](https://cursor.com/), [GitHub Copilot](https://github.com/features/copilot) and [Claude Code](https://claude.com/product/claude-code) and [Google Gemini CLI](https://github.com/google-gemini/gemini-cli) to use the Shopify Dev MCP.

For more information on the Shopify Dev MCP please read [the documentation](https://shopify.dev/docs/apps/build/devmcp).

## Deployment

### Application Storage

This template uses [Prisma](https://www.prisma.io/) to store session data, by default using an [SQLite](https://www.sqlite.org/index.html) database.
The database is defined as a Prisma schema in `prisma/schema.prisma`.

This use of SQLite works in production if your app runs as a single instance.
The database that works best for you depends on the data your app needs and how it is queried.
Here’s a short list of databases providers that provide a free tier to get started:

| Database   | Type             | Hosters                                                                                                                                                                                                                                    |
| ---------- | ---------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------ |
| MySQL      | SQL              | [Digital Ocean](https://www.digitalocean.com/products/managed-databases-mysql), [Planet Scale](https://planetscale.com/), [Amazon Aurora](https://aws.amazon.com/rds/aurora/), [Google Cloud SQL](https://cloud.google.com/sql/docs/mysql) |
| PostgreSQL | SQL              | [Digital Ocean](https://www.digitalocean.com/products/managed-databases-postgresql), [Amazon Aurora](https://aws.amazon.com/rds/aurora/), [Google Cloud SQL](https://cloud.google.com/sql/docs/postgres)                                   |
| Redis      | Key-value        | [Digital Ocean](https://www.digitalocean.com/products/managed-databases-redis), [Amazon MemoryDB](https://aws.amazon.com/memorydb/)                                                                                                        |
| MongoDB    | NoSQL / Document | [Digital Ocean](https://www.digitalocean.com/products/managed-databases-mongodb), [MongoDB Atlas](https://www.mongodb.com/atlas/database)                                                                                                  |

To use one of these, you can use a different [datasource provider](https://www.prisma.io/docs/reference/api-reference/prisma-schema-reference#datasource) in your `schema.prisma` file, or a different [SessionStorage adapter package](https://github.com/Shopify/shopify-api-js/blob/main/packages/shopify-api/docs/guides/session-storage.md).

## Gotchas / Troubleshooting

### Database tables don't exist

If you get an error like:

```
The table `main.Session` does not exist in the current database.
```

Create the database for Prisma. Run the `setup` script in `package.json` using `npm`, `yarn` or `pnpm`.

### Navigating/redirecting breaks an embedded app

Embedded apps must maintain the user session, which can be tricky inside an iFrame. To avoid issues:

1. Use `Link` from `react-router` or `@shopify/polaris`. Do not use `<a>`.
2. Use `redirect` returned from `authenticate.admin`. Do not use `redirect` from `react-router`
3. Use `useSubmit` from `react-router`.

This only applies if your app is embedded, which it will be by default.

### Webhooks: shop-specific webhook subscriptions aren't updated

If you are registering webhooks in the `afterAuth` hook, using `shopify.registerWebhooks`, you may find that your subscriptions aren't being updated.

Instead of using the `afterAuth` hook declare app-specific webhooks in the `shopify.app.toml` file. This approach is easier since Shopify will automatically sync changes every time you run `deploy` (e.g: `npm run deploy`). Please read these guides to understand more:

1. [app-specific vs shop-specific webhooks](https://shopify.dev/docs/apps/build/webhooks/subscribe#app-specific-subscriptions)
2. [Create a subscription tutorial](https://shopify.dev/docs/apps/build/webhooks/subscribe/get-started?deliveryMethod=https)

If you do need shop-specific webhooks, keep in mind that the package calls `afterAuth` in 2 scenarios:

- After installing the app
- When an access token expires

During normal development, the app won't need to re-authenticate most of the time, so shop-specific subscriptions aren't updated. To force your app to update the subscriptions, uninstall and reinstall the app. Revisiting the app will call the `afterAuth` hook.

### Webhooks: Admin created webhook failing HMAC validation

Webhooks subscriptions created in the [Shopify admin](https://help.shopify.com/en/manual/orders/notifications/webhooks) will fail HMAC validation. This is because the webhook payload is not signed with your app's secret key.

The recommended solution is to use [app-specific webhooks](https://shopify.dev/docs/apps/build/webhooks/subscribe#app-specific-subscriptions) defined in your toml file instead. Test your webhooks by triggering events manually in the Shopify admin(e.g. Updating the product title to trigger a `PRODUCTS_UPDATE`).

### Webhooks: Admin object undefined on webhook events triggered by the CLI

When you trigger a webhook event using the Shopify CLI, the `admin` object will be `undefined`. This is because the CLI triggers an event with a valid, but non-existent, shop. The `admin` object is only available when the webhook is triggered by a shop that has installed the app. This is expected.

Webhooks triggered by the CLI are intended for initial experimentation testing of your webhook configuration. For more information on how to test your webhooks, see the [Shopify CLI documentation](https://shopify.dev/docs/apps/tools/cli/commands#webhook-trigger).

### Incorrect GraphQL Hints

By default the [graphql.vscode-graphql](https://marketplace.visualstudio.com/items?itemName=GraphQL.vscode-graphql) extension for will assume that GraphQL queries or mutations are for the [Shopify Admin API](https://shopify.dev/docs/api/admin). This is a sensible default, but it may not be true if:

1. You use another Shopify API such as the storefront API.
2. You use a third party GraphQL API.

If so, please update [.graphqlrc.ts](https://github.com/Shopify/shopify-app-template-react-router/blob/main/.graphqlrc.ts).

### Using Defer & await for streaming responses

By default the CLI uses a cloudflare tunnel. Unfortunately cloudflare tunnels wait for the Response stream to finish, then sends one chunk. This will not affect production.

To test [streaming using await](https://reactrouter.com/api/components/Await#await) during local development we recommend [localhost based development](https://shopify.dev/docs/apps/build/cli-for-apps/networking-options#localhost-based-development).

### "nbf" claim timestamp check failed

This is because a JWT token is expired. If you are consistently getting this error, it could be that the clock on your machine is not in sync with the server. To fix this ensure you have enabled "Set time and date automatically" in the "Date and Time" settings on your computer.

### Using MongoDB and Prisma

If you choose to use MongoDB with Prisma, there are some gotchas in Prisma's MongoDB support to be aware of. Please see the [Prisma SessionStorage README](https://www.npmjs.com/package/@shopify/shopify-app-session-storage-prisma#mongodb).

### Unable to require(`C:\...\query_engine-windows.dll.node`).

Unable to require(`C:\...\query_engine-windows.dll.node`).
The Prisma engines do not seem to be compatible with your system.

query_engine-windows.dll.node is not a valid Win32 application.

**Fix:** Set the environment variable:

```shell
PRISMA_CLIENT_ENGINE_TYPE=binary
```

This forces Prisma to use the binary engine mode, which runs the query engine as a separate process and can work via emulation on Windows ARM64.

import { reactRouter } from "@react-router/dev/vite";
import { defineConfig } from "vite";
import tsconfigPaths from "vite-tsconfig-paths";

if (
  process.env.HOST &&
  (!process.env.SHOPIFY_APP_URL ||
    process.env.SHOPIFY_APP_URL === process.env.HOST)
) {
  process.env.SHOPIFY_APP_URL = process.env.HOST;
  delete process.env.HOST;
}

const host = new URL(process.env.SHOPIFY_APP_URL || "http://localhost")
  .hostname;
let hmrConfig;

if (host === "localhost") {
  hmrConfig = {
    protocol: "ws",
    host: "localhost",
    port: 64999,
    clientPort: 64999,
  };
} else {
  hmrConfig = {
    protocol: "wss",
    host: host,
    port: parseInt(process.env.FRONTEND_PORT) || 8002,
    clientPort: 443,
  };
}

export default defineConfig({
  server: {
    allowedHosts: [host],
    cors: {
      preflightContinue: true,
    },
    port: Number(process.env.PORT || 3000),
    hmr: hmrConfig,
    fs: {
      allow: ["app", "node_modules"],
    },
  },
  plugins: [reactRouter(), tsconfigPaths()],
  build: {
    assetsInlineLimit: 0,
  },
  optimizeDeps: {
    include: ["@shopify/app-bridge-react"],
  },
});
